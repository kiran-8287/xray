// ─────────────────────────────────────────────
// HELPERS
// ─────────────────────────────────────────────
const border1 = (col=C.border) => ({ style: BorderStyle.SINGLE, size: 4, color: col });
const noBorder = () => ({ style: BorderStyle.NONE, size: 0, color: 'FFFFFF' });
const allBorders = (col=C.border) => ({ top: border1(col), bottom: border1(col), left: border1(col), right: border1(col) });
const noBorders = () => ({ top: noBorder(), bottom: noBorder(), left: noBorder(), right: noBorder() });

const cell = (children, w, opts={}) => new TableCell({
  children,
  width: { size: w, type: WidthType.DXA },
  borders: opts.borders || allBorders(opts.borderColor || C.border),
  shading: opts.fill ? { fill: opts.fill, type: ShadingType.CLEAR } : undefined,
  margins: { top: 80, bottom: 80, left: 140, right: 140 },
  verticalAlign: opts.valign || VerticalAlign.CENTER,
});

const row = (cells, opts={}) => new TableRow({ children: cells, tableHeader: opts.header || false });

const para = (runs, opts={}) => new Paragraph({
  children: Array.isArray(runs) ? runs : [runs],
  alignment: opts.align || AlignmentType.LEFT,
  spacing: { before: opts.before || 0, after: opts.after || 80 },
  numbering: opts.numbering,
  border: opts.border,
});

const run = (text, opts={}) => new TextRun({
  text,
  font: 'Arial',
  size: opts.size || 20,
  bold: opts.bold || false,
  italics: opts.italic || false,
  color: opts.color || C.ink,
});

// ─────────────────────────────────────────────
// SWATCH TABLE
// ─────────────────────────────────────────────
function makeSwatchTable(swatches, totalW=9360) {
  const colW = Math.floor(totalW / swatches.length);
  return new Table({
    width: { size: totalW, type: WidthType.DXA },
    columnWidths: swatches.map(() => colW),
    rows: [
      row(swatches.map(s => cell(
        [para(run('', { size: 10 }))],
        colW,
        { fill: s.hex, borders: allBorders('CCCCCC') }
      ))),
      row(swatches.map(s => cell(
        [para(run(s.name, { size: 17, bold: true }), { align: AlignmentType.CENTER }),
         para(run('#' + s.hex, { size: 16, color: C.muted }), { align: AlignmentType.CENTER, after: 0 })],
        colW,
        { fill: C.panel, borders: allBorders(C.border) }
      ))),
    ],
  });
}

// ─────────────────────────────────────────────
// SPEC TABLE (used for prop lists)
// ─────────────────────────────────────────────
function specTable(rows_data, colWidths=[3000, 6360]) {
  const total = colWidths.reduce((a,b)=>a+b, 0);
  return new Table({
    width: { size: total, type: WidthType.DXA },
    columnWidths: colWidths,
    rows: [
      row([
        cell([para(run('Property', { size: 18, bold: true, color: C.panel }))], colWidths[0], { fill: C.ink }),
        cell([para(run('Value / Class String', { size: 18, bold: true, color: C.panel }))], colWidths[1], { fill: C.ink }),
      ], { header: true }),
      ...rows_data.map((r, i) => row([
        cell([para(run(r[0], { size: 18, bold: true, color: C.ink }))], colWidths[0], { fill: i%2===0 ? C.surface : C.panel }),
        cell([para(run(r[1], { size: 18, color: C.accentSky, italic: r[2] || false }))], colWidths[1], { fill: i%2===0 ? C.surface : C.panel }),
      ]))
    ],
  });
}

// ─────────────────────────────────────────────
// 2-COL LAYOUT TABLE
// ─────────────────────────────────────────────
function twoCol(left, right, lW=4500, rW=4860) {
  return new Table({
    width: { size: lW+rW, type: WidthType.DXA },
    columnWidths: [lW, rW],
    rows: [row([
      cell(left, lW, { borders: noBorders() }),
      cell(right, rW, { borders: noBorders() }),
    ])],
  });
}

// ─────────────────────────────────────────────
// CALLOUT BOX
// ─────────────────────────────────────────────
function callout(label, text, bg, textCol, bdrCol) {
  return new Table({
    width: { size: 9360, type: WidthType.DXA },
    columnWidths: [9360],
    rows: [row([cell([
      para(run('▎  ' + label, { size: 18, bold: true, color: textCol }), { after: 60 }),
      para(run(text, { size: 18, color: textCol }), { after: 0 }),
    ], 9360, { fill: bg, borderColor: bdrCol })])],
  });
}

  ];
}
